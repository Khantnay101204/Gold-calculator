import React, { useState, useMemo } from 'react';
import { RotateCcw, Scale, Coins, Info, Calendar } from 'lucide-react';

export default function App() {
  // Navigation Tab State: 'gold' | 'pawn'
  const [activeTab, setActiveTab] = useState('gold');

  // ==========================================
  // PAGE 1: GOLD CALCULATOR STATE
  // ==========================================
  const [marketPrice, setMarketPrice] = useState('9900000');

  // Product Weight (အထည်ချိန်)
  const [prodKyat, setProdKyat] = useState('');
  const [prodPae, setProdPae] = useState('');
  const [prodYwe, setProdYwe] = useState('');

  // Wastage Weight (အလျော့တွက်)
  const [wasteKyat, setWasteKyat] = useState('');
  const [wastePae, setWastePae] = useState('');
  const [wasteYwe, setWasteYwe] = useState('');

  // Labor Fee (လက်ခ)
  const [laborCost, setLaborCost] = useState('');

  const handleResetGold = () => {
    setProdKyat('');
    setProdPae('');
    setProdYwe('');
    setWasteKyat('');
    setWastePae('');
    setWasteYwe('');
    setLaborCost('');
  };

  // 1 Kyat = 128 Ywe, 1 Pae = 8 Ywe
  const calculateTotalYwe = (kStr, pStr, yStr) => {
    const k = parseFloat(kStr) || 0;
    const p = parseFloat(pStr) || 0;
    const y = parseFloat(yStr) || 0;
    return k * 128 + p * 8 + y;
  };

  const goldCalculations = useMemo(() => {
    const pricePerKyat = parseFloat(marketPrice) || 0;
    const pricePerYwe = pricePerKyat / 128;

    const prodYweTotal = calculateTotalYwe(prodKyat, prodPae, prodYwe);
    const wasteYweTotal = calculateTotalYwe(wasteKyat, wastePae, wasteYwe);

    const productPrice = Math.round(prodYweTotal * pricePerYwe);
    const wastagePrice = Math.round(wasteYweTotal * pricePerYwe);
    const goldTotal = productPrice + wastagePrice;
    const labor = parseFloat(laborCost) || 0;
    const grandTotal = goldTotal + labor;

    return { productPrice, wastagePrice, goldTotal, labor, grandTotal };
  }, [
    marketPrice,
    prodKyat,
    prodPae,
    prodYwe,
    wasteKyat,
    wastePae,
    wasteYwe,
    laborCost,
  ]);

  // ==========================================
  // PAGE 2: PAWN / POUNING CALCULATOR STATE
  // ==========================================
  const [pawnAmount, setPawnAmount] = useState('1000000');
  const [interestRate, setInterestRate] = useState('3'); // Default 3%

  // Today's date defaults for Start / End dates
  const today = new Date();
  const nextMonth = new Date();
  nextMonth.setMonth(today.getMonth() + 1);

  // Start Date Inputs
  const [startDay, setStartDay] = useState(today.getDate().toString());
  const [startMonth, setStartMonth] = useState((today.getMonth() + 1).toString());
  const [startYear, setStartYear] = useState(today.getFullYear().toString());

  // End Date Inputs
  const [endDay, setEndDay] = useState(nextMonth.getDate().toString());
  const [endMonth, setEndMonth] = useState((nextMonth.getMonth() + 1).toString());
  const [endYear, setEndYear] = useState(nextMonth.getFullYear().toString());

  const handleResetPawn = () => {
    setPawnAmount('1000000');
    setInterestRate('3');
    setStartDay('1');
    setStartMonth('2');
    setStartYear('2026');
    setEndDay('1');
    setEndMonth('3');
    setEndYear('2026');
  };

  // Duration & Interest Calculation Logic
  const pawnCalculations = useMemo(() => {
    const principal = parseFloat(pawnAmount) || 0;
    const rate = parseFloat(interestRate) || 0;

    const sD = parseInt(startDay, 10) || 1;
    const sM = (parseInt(startMonth, 10) || 1) - 1; // 0-indexed month
    const sY = parseInt(startYear, 10) || 2026;

    const eD = parseInt(endDay, 10) || 1;
    const eM = (parseInt(endMonth, 10) || 1) - 1;
    const eY = parseInt(endYear, 10) || 2026;

    const startDate = new Date(sY, sM, sD);
    const endDate = new Date(eY, eM, eD);

    // If date input is invalid or end date < start date
    if (isNaN(startDate.getTime()) || isNaN(endDate.getTime()) || endDate < startDate) {
      return {
        principal,
        rate,
        monthsElapsed: 0,
        daysElapsed: 0,
        chargeableMonths: 0,
        monthlyInterest: 0,
        totalInterest: 0,
        totalRepay: principal,
        ruleText: 'ရက်စွဲ စစ်ဆေးပါ။ (ရွေးယူသည့်ရက်သည် စတင်သည့်ရက်ထက် စော၍မရပါ)',
        isValid: false,
      };
    }

    // Calculate calendar difference (Full Months + Extra Days)
    let months = (endDate.getFullYear() - startDate.getFullYear()) * 12 + (endDate.getMonth() - startDate.getMonth());
    let days = endDate.getDate() - startDate.getDate();

    if (days < 0) {
      months -= 1;
      // Get days in previous month
      const prevMonthLastDay = new Date(endDate.getFullYear(), endDate.getMonth(), 0).getDate();
      days += prevMonthLastDay;
    }

    let chargeableMonths = 1;
    let ruleText = '';

    // RULE 1: Within Month 1 (months === 0 and days > 0) -> strictly 1.0 month (0.5 not allowed in month 1)
    if (months === 0) {
      chargeableMonths = 1.0;
      if (days === 0) {
        ruleText = '၀ လ + ၀ ရက် = ၁ လ (အနည်းဆုံး ၁ လ စည်းမျဉ်း)';
      } else {
        ruleText = `၀ လ + ${days} ရက် = ၁ လ (ပထမလတွင် ၀.၅ လ မရှိပါ)`;
      }
    } else {
      // RULE 2: Months >= 1
      let extra = 0;
      if (days > 0 && days <= 15) {
        extra = 0.5;
        ruleText = `${months} လ + ${days} ရက် = ${months + 0.5} လ (၁၅ ရက်အထိ +၀.၅ လ)`;
      } else if (days > 15) {
        extra = 1.0;
        ruleText = `${months} လ + ${days} ရက် = ${months + 1.0} လ (၁၅ ရက်ကျော် +၁.၀ လ)`;
      } else {
        ruleText = `${months} လ ကွက်တိ = ${months} လ`;
      }
      chargeableMonths = months + extra;
    }

    const monthlyInterest = Math.round(principal * (rate / 100));
    const totalInterest = Math.round(monthlyInterest * chargeableMonths);
    const totalRepay = principal + totalInterest;

    return {
      principal,
      rate,
      monthsElapsed: months,
      daysElapsed: days,
      chargeableMonths,
      monthlyInterest,
      totalInterest,
      totalRepay,
      ruleText,
      isValid: true,
    };
  }, [pawnAmount, interestRate, startDay, startMonth, startYear, endDay, endMonth, endYear]);

  // Number Formatter
  const formatMoney = (num) => {
    return new Intl.NumberFormat('en-US').format(num || 0);
  };

  return (
    <div className="min-h-screen bg-slate-100 flex flex-col items-center justify-start p-3 sm:p-6 font-sans text-slate-800">
      <div className="w-full max-w-md bg-white rounded-2xl p-5 sm:p-6 shadow-md border border-slate-200/80 space-y-5">
        
        {/* TOP TAB NAVIGATION */}
        <div className="grid grid-cols-2 gap-2 p-1.5 bg-slate-100 rounded-xl border border-slate-200/60">
          <button
            onClick={() => setActiveTab('gold')}
            type="button"
            className={`flex items-center justify-center gap-2 py-2.5 px-3 rounded-lg font-bold text-sm sm:text-base transition ${
              activeTab === 'gold'
                ? 'bg-amber-500 text-white shadow-sm'
                : 'text-slate-600 hover:text-slate-900 hover:bg-slate-200/50'
            }`}
          >
            <Coins className="w-4 h-4 sm:w-5 sm:h-5" />
            <span>ရွှေတွက်ရန်</span>
          </button>

          <button
            onClick={() => setActiveTab('pawn')}
            type="button"
            className={`flex items-center justify-center gap-2 py-2.5 px-3 rounded-lg font-bold text-sm sm:text-base transition ${
              activeTab === 'pawn'
                ? 'bg-amber-500 text-white shadow-sm'
                : 'text-slate-600 hover:text-slate-900 hover:bg-slate-200/50'
            }`}
          >
            <Scale className="w-4 h-4 sm:w-5 sm:h-5" />
            <span>အပေါင်တွက်ရန်</span>
          </button>
        </div>

        {/* ========================================================
            TAB 1: GOLD CALCULATOR
            ======================================================== */}
        {activeTab === 'gold' && (
          <div className="space-y-5">
            {/* Header & Reset */}
            <div className="flex items-center justify-between border-b border-slate-100 pb-3">
              <h2 className="text-base font-bold text-slate-800 flex items-center gap-2">
                <Coins className="w-5 h-5 text-amber-500" />
                ရွှေကျသင့်ငွေ တွက်ချက်ရန်
              </h2>
              <button
                onClick={handleResetGold}
                type="button"
                className="flex items-center gap-1 text-xs font-medium text-slate-500 hover:text-amber-600 bg-slate-50 hover:bg-amber-50 border border-slate-200 hover:border-amber-200 px-2.5 py-1.5 rounded-lg transition"
              >
                <RotateCcw className="w-3.5 h-3.5" />
                <span>ပြန်စမည်</span>
              </button>
            </div>

            {/* 1. ပေါက်ဈေး (၁ ကျပ်) Input */}
            <div className="space-y-1.5">
              <label className="block text-sm sm:text-base font-semibold text-slate-700">
                ပေါက်ဈေး (၁ ကျပ်):
              </label>
              <input
                type="number"
                value={marketPrice}
                onChange={(e) => setMarketPrice(e.target.value)}
                placeholder="9900000"
                className="w-full text-xl sm:text-2xl font-medium tracking-wide text-slate-900 px-4 py-3 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-300"
              />
            </div>

            {/* 2. အထည်ချိန် (Product Weight) Inputs */}
            <div className="space-y-1.5">
              <label className="block text-sm font-semibold text-slate-700">
                အထည်ချိန်:
              </label>
              <div className="grid grid-cols-3 gap-2.5">
                <input
                  type="number"
                  min="0"
                  value={prodKyat}
                  onChange={(e) => setProdKyat(e.target.value)}
                  placeholder="ကျပ်"
                  className="w-full text-center text-lg font-medium text-slate-800 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="0"
                  max="15"
                  value={prodPae}
                  onChange={(e) => setProdPae(e.target.value)}
                  placeholder="ပဲ"
                  className="w-full text-center text-lg font-medium text-slate-800 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="0"
                  max="7.9"
                  step="0.1"
                  value={prodYwe}
                  onChange={(e) => setProdYwe(e.target.value)}
                  placeholder="ရွေး"
                  className="w-full text-center text-lg font-medium text-slate-800 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
              </div>
            </div>

            {/* 3. အလျော့တွက် (Wastage Weight) Inputs */}
            <div className="space-y-1.5">
              <label className="block text-sm font-semibold text-slate-700">
                အလျော့တွက်:
              </label>
              <div className="grid grid-cols-3 gap-2.5">
                <input
                  type="number"
                  min="0"
                  value={wasteKyat}
                  onChange={(e) => setWasteKyat(e.target.value)}
                  placeholder="ကျပ်"
                  className="w-full text-center text-lg font-medium text-slate-800 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="0"
                  max="15"
                  value={wastePae}
                  onChange={(e) => setWastePae(e.target.value)}
                  placeholder="ပဲ"
                  className="w-full text-center text-lg font-medium text-slate-800 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="0"
                  max="7.9"
                  step="0.1"
                  value={wasteYwe}
                  onChange={(e) => setWasteYwe(e.target.value)}
                  placeholder="ရွေး"
                  className="w-full text-center text-lg font-medium text-slate-800 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
              </div>
            </div>

            {/* 4. လက်ခ (Labor Cost) Input */}
            <div className="space-y-1.5">
              <label className="block text-sm font-semibold text-slate-700">
                လက်ခ (ကျပ်):
              </label>
              <input
                type="number"
                min="0"
                value={laborCost}
                onChange={(e) => setLaborCost(e.target.value)}
                placeholder="လက်ခ ထည့်ပါ"
                className="w-full text-lg font-medium text-slate-800 px-4 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
              />
            </div>

            {/* 5. Result Display Box */}
            <div className="bg-sky-50 border border-sky-100 rounded-xl p-4 space-y-2.5">
              <div className="flex items-center justify-between text-sm text-slate-600">
                <span>အထည်ဖိုး:</span>
                <span className="font-semibold text-slate-800">
                  {formatMoney(goldCalculations.productPrice)} Ks
                </span>
              </div>

              <div className="flex items-center justify-between text-sm text-slate-600">
                <span>အလျော့တွက်ဖိုး:</span>
                <span className="font-semibold text-slate-800">
                  {formatMoney(goldCalculations.wastagePrice)} Ks
                </span>
              </div>

              <div className="flex items-center justify-between text-sm text-slate-600 border-t border-sky-200/60 pt-2">
                <span>ရွှေကျသင့်ငွေ:</span>
                <span className="font-semibold text-slate-900">
                  {formatMoney(goldCalculations.goldTotal)} Ks
                </span>
              </div>

              <div className="flex items-center justify-between text-sm text-slate-600">
                <span>လက်ခ:</span>
                <span className="font-semibold text-slate-800">
                  {formatMoney(goldCalculations.labor)} Ks
                </span>
              </div>

              <div className="flex items-center justify-between border-t border-sky-200 pt-2.5 mt-1">
                <span className="text-base sm:text-lg font-bold text-slate-800">
                  စုစုပေါင်းကျသင့်ငွေ:
                </span>
                <span className="text-xl sm:text-2xl font-bold text-amber-700 tracking-wide">
                  {formatMoney(goldCalculations.grandTotal)}{' '}
                  <span className="text-amber-600 font-semibold text-lg sm:text-xl">
                    Ks
                  </span>
                </span>
              </div>
            </div>
          </div>
        )}

        {/* ========================================================
            TAB 2: PAWN / POUNING CALCULATOR
            ======================================================== */}
        {activeTab === 'pawn' && (
          <div className="space-y-5">
            {/* Header & Reset */}
            <div className="flex items-center justify-between border-b border-slate-100 pb-3">
              <h2 className="text-base font-bold text-slate-800 flex items-center gap-2">
                <Scale className="w-5 h-5 text-amber-500" />
                အပေါင်/အတိုး တွက်ချက်ရန်
              </h2>
              <button
                onClick={handleResetPawn}
                type="button"
                className="flex items-center gap-1 text-xs font-medium text-slate-500 hover:text-amber-600 bg-slate-50 hover:bg-amber-50 border border-slate-200 hover:border-amber-200 px-2.5 py-1.5 rounded-lg transition"
              >
                <RotateCcw className="w-3.5 h-3.5" />
                <span>ပြန်စမည်</span>
              </button>
            </div>

            {/* 1. ပေါင်နှံငွေ (Pawn Principal) & အတိုးနှုန်း (Interest Rate) */}
            <div className="grid grid-cols-3 gap-3">
              <div className="col-span-2 space-y-1.5">
                <label className="block text-sm font-semibold text-slate-700">
                  ပေါင်နှံငွေ (ကျပ်):
                </label>
                <input
                  type="number"
                  value={pawnAmount}
                  onChange={(e) => setPawnAmount(e.target.value)}
                  placeholder="1000000"
                  className="w-full text-lg sm:text-xl font-medium text-slate-900 px-3.5 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition"
                />
              </div>

              <div className="col-span-1 space-y-1.5">
                <label className="block text-sm font-semibold text-slate-700">
                  အတိုးနှုန်း %:
                </label>
                <div className="relative">
                  <input
                    type="number"
                    step="0.1"
                    value={interestRate}
                    onChange={(e) => setInterestRate(e.target.value)}
                    className="w-full text-lg font-medium text-center text-slate-900 py-2.5 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition"
                  />
                  <span className="absolute right-2 top-2.5 text-xs text-slate-400 font-medium">
                    /လ
                  </span>
                </div>
              </div>
            </div>

            {/* 2. START DATE (စတင်သည့်ရက် - DD / MM / YYYY) */}
            <div className="space-y-1.5">
              <label className="block text-sm font-semibold text-slate-700 flex items-center gap-1.5">
                <Calendar className="w-4 h-4 text-amber-500" />
                စတင်သည့်ရက် (ရက် / လ / နှစ်):
              </label>
              <div className="grid grid-cols-3 gap-2">
                <input
                  type="number"
                  min="1"
                  max="31"
                  value={startDay}
                  onChange={(e) => setStartDay(e.target.value)}
                  placeholder="ရက် (DD)"
                  className="w-full text-center text-base font-medium text-slate-800 py-2 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="1"
                  max="12"
                  value={startMonth}
                  onChange={(e) => setStartMonth(e.target.value)}
                  placeholder="လ (MM)"
                  className="w-full text-center text-base font-medium text-slate-800 py-2 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="2000"
                  max="2100"
                  value={startYear}
                  onChange={(e) => setStartYear(e.target.value)}
                  placeholder="နှစ် (YYYY)"
                  className="w-full text-center text-base font-medium text-slate-800 py-2 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
              </div>
            </div>

            {/* 3. END DATE (ရွေးယူသည့်ရက် - DD / MM / YYYY) */}
            <div className="space-y-1.5">
              <label className="block text-sm font-semibold text-slate-700 flex items-center gap-1.5">
                <Calendar className="w-4 h-4 text-emerald-600" />
                ရွေးယူသည့်ရက် (ရက် / လ / နှစ်):
              </label>
              <div className="grid grid-cols-3 gap-2">
                <input
                  type="number"
                  min="1"
                  max="31"
                  value={endDay}
                  onChange={(e) => setEndDay(e.target.value)}
                  placeholder="ရက် (DD)"
                  className="w-full text-center text-base font-medium text-slate-800 py-2 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="1"
                  max="12"
                  value={endMonth}
                  onChange={(e) => setEndMonth(e.target.value)}
                  placeholder="လ (MM)"
                  className="w-full text-center text-base font-medium text-slate-800 py-2 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
                <input
                  type="number"
                  min="2000"
                  max="2100"
                  value={endYear}
                  onChange={(e) => setEndYear(e.target.value)}
                  placeholder="နှစ် (YYYY)"
                  className="w-full text-center text-base font-medium text-slate-800 py-2 bg-white border border-slate-300 rounded-xl focus:outline-none focus:ring-2 focus:ring-amber-400 focus:border-transparent transition placeholder-slate-400"
                />
              </div>
            </div>

            {/* Interest Rules Helper Note */}
            <div className="flex items-start gap-2 p-2.5 bg-amber-50/80 border border-amber-200/60 rounded-xl text-xs text-amber-900 leading-relaxed">
              <Info className="w-4 h-4 text-amber-600 shrink-0 mt-0.5" />
              <div>
                <strong>အတိုးစနစ် စည်းမျဉ်း:</strong> ပထမ ၁ လအတွင်း (၀.၅ လ) မရှိပါ။ ၁ လ ပြည့်အဖြစ် တွက်သည်။ ပထမလကျော်မှသာ ၁၅ ရက်အထိ (+၀.၅ လ)၊ ၁၅ ရက်ကျော်ပါက (+၁.၀ လ) တွက်ချက်ပါသည်။
              </div>
            </div>

            {/* 4. Result Display Box */}
            <div className="bg-sky-50 border border-sky-100 rounded-xl p-4 space-y-2.5">
